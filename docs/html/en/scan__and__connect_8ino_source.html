100 while (!g_codex_pad_client.ScanAndConnect(kExpectedButtonMask)) {

118void RssiMonitor() {

121 if (s_last_time == 0 || millis() - s_last_time > 1000) {

125 CLOGI("rssi: %" PRId32 " dBm, | [#][#][#][#] | strong", rssi);

151void loop() {
152 // ==========================================================================
153 // 🔴 CRITICAL: Call Update() as frequently as possible in loop()
154 // ==========================================================================
155 // • Update() processes incoming Bluetooth packets from the CodexPad
156 // • Any delay(...) or long blocking code WILL cause:
157 // - Packet loss
158 // - Input lag
159 // - Unstable connection
160 //
161 // • For real-time control, call Update() every loop iteration
162 // without any blocking operations
163 //
164 // 🔴【重要】Update() 必须在 loop() 中尽可能高频调用
165 // • Update() 负责处理来自 CodexPad 的蓝牙数据包
166 // • 任何形式的 delay 或阻塞代码都会导致：
167 // - 数据丢失
168 // - 响应延迟
169 // - 连接不稳定
170 //
171 // • 实时控制应用中，必须每轮循环都调用 Update()，不可阻塞
172 // ==========================================================================
173 const gamepad::input::Tracker& it = g_codex_pad_client.Update();
174 // ==========================================================================
175 // Tracker: Gamepad Input Snapshot & Change Engine
176 // ==========================================================================
177 // • Returned by Update()
178 // • Maintains previous and current input snapshots
179 // • Enables edge detection and delta detection
180 //
181 // Tracker 由 Update() 返回
182 // 内部保存上一帧和当前帧的输入数据
183 // 支持边沿检测和差值检测
184 //
185 // 📚 https://codexpad.github.io/gamepad_input_arduino_lib/
186 // ==========================================================================
187
188 if (!g_codex_pad_client.is_connected()) {
189 CLOGW("Disconnected, start to reconnect");
190 Connect();
191 return;
192 }
193
194 RssiMonitor();
195
196 // ==========================================================================
197 // 🟢 Button State Change Detection (Edge-Based)
198 // ==========================================================================
199 // • pressed() → button was just pressed (released → pressed)
200 // • released() → button was just released (pressed → released)
201 // • holding() → button remains pressed across frames
202 //
203 // These APIs are *frame-differential* and rely on Update() frequency.
204 // Ideal for UI navigation, action triggers, and avoiding repeat firing.
205 //
206 // • pressed() → 按钮刚被按下（弹起 → 按下）
207 // • released() → 按钮刚被释放（按下 → 弹起）
208 // • holding() → 按钮在两帧之间持续按下
209 //
210 // 这些接口是“帧间差分”的，依赖于 Update() 的高频调用
211 // 非常适合 UI 导航、动作触发和防止长按连发
212 // ==========================================================================
213 static std::map<Button, uint64_t> s_button_last_time;
214
215 for (const auto& [button, name] : kButtonNames) {
216 if (it.pressed(button)) { // button was just pressed
217 CLOGI("Button mask: 0x%08" PRIX32 ", button %s: pressed", it.raw().buttons, name.c_str());
218 s_button_last_time[button] = millis();
219 } else if (it.released(button)) { // button was just released
220 CLOGI("Button mask: 0x%08" PRIX32 ", button %s: released", it.raw().buttons, name.c_str());
221 s_button_last_time[button] = millis();
222 } else if (it.holding(button)) { // button remains pressed
223 if (s_button_last_time.find(button) == s_button_last_time.end() || s_button_last_time[button] == 0 ||
224 millis() - s_button_last_time[button] > 100) {
225 CLOGI("Button mask: 0x%08" PRIX32 ", button %s: holding", it.raw().buttons, name.c_str());
226 s_button_last_time[button] = millis();
227 }
228 }
229 }
230
231 // ==========================================================================
232 // 🟢 Joystick Axis Change Detection (Threshold-Based Filtering)
233 // ==========================================================================
234 // • AxisChanged() detects significant changes between frames
235 // • Uses a threshold to filter out noise and minor jitter
236 // • Only reports movement when change ≥ threshold
237 //
238 // • AxisChanged() 用于检测摇杆轴值的有效变化
239 // • 使用阈值过滤微小抖动和噪声
240 // • 只有当变化幅度 ≥ 阈值时才视为有效移动
241 // ==========================================================================
242 constexpr uint8_t kAxisValueChangeThreshold = 5;
243
244 if (it.AxisChanged(Axis::kLeftStickX, kAxisValueChangeThreshold) ||
245 it.AxisChanged(Axis::kLeftStickY, kAxisValueChangeThreshold) ||
246 it.AxisChanged(Axis::kRightStickX, kAxisValueChangeThreshold) ||
247 it.AxisChanged(Axis::kRightStickY, kAxisValueChangeThreshold)) {
248 CLOGI("L(X: %3" PRIu8 ", Y:%3" PRIu8 "), R(X: %3" PRIu8 ", Y: %3" PRIu8 ")", it[Axis::kLeftStickX], it[Axis::kLeftStickY],
249 it[Axis::kRightStickX], it[Axis::kRightStickY]);
250 }
