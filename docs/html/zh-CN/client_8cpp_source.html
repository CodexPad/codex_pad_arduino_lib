144bool Client::Connect(const NimBLEAddress& address, bool async_connect, const uint32_t timeout_ms) {
145 Reset();
146 assert(ble_client_ == nullptr);
147 ble_client_ = NimBLEDevice::createClient(address);
148 ble_client_->setConnectTimeout(timeout_ms);
149 ble_client_->setConnectionParams(kConnectionParamMinInterval, kConnectionParamMaxInterval, kConnectionParamLatency,
150 kConnectionParamSupervisionTimeout);
151 ble_client_->setClientCallbacks(this, false);
152 auto ret = ble_client_->connect(true, async_connect, true);
153
154 if (!ret || !ble_client_->isConnected()) {
155 goto FAILED;
156 }
157
158 remote_device_name_ = ble_client_->getValue(kGapServiceUuid, kGapDeviceNameUuid);
159 remote_model_number_ = ble_client_->getValue(kDeviceInfoServiceUuid, kModelNumberCharacteristicUuid);
160 {
161 auto firmware_revision = ble_client_->getValue(kDeviceInfoServiceUuid, kFirmwareRevisionCharacteristicUuid);
162 if (firmware_revision.length() == remote_firmware_version_.size()) {
163 memcpy(remote_firmware_version_.data(), firmware_revision.data(), firmware_revision.length());
164 }
165 }
166
167 {
168 auto remote_service = ble_client_->getService(kInputsServiceUuid);
169 if (remote_service == nullptr) {
170 goto FAILED;
171 }
172
173 auto remote_characteristic = remote_service->getCharacteristic(kInputsCharacteristicUuid);
174 if (remote_characteristic == nullptr) {
175 goto FAILED;
176 }
177
178 if (!remote_characteristic->canNotify()) {
179 goto FAILED;
180 }
181
182 if (!remote_characteristic->subscribe(true, std::bind(&Client::OnNotify, this, std::placeholders::_1, std::placeholders::_2,
183 std::placeholders::_3, std::placeholders::_4))) {
184 goto FAILED;
185 }
186 }
187
188 ble_client_->updatePhy(BLE_GAP_LE_PHY_CODED_MASK, BLE_GAP_LE_PHY_CODED_MASK, BLE_GAP_LE_PHY_CODED_S8);
189
190 return ret;
191
192FAILED:
193 Reset();
194 return false;
195}
